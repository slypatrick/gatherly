/* Gatherly V32 — authorized cross-board batch health repair */
async function g32BatchHealth(req,user){
 if(req.method!=="POST")return fail("Method not allowed",405);
 if(!user)return fail("Sign in to repair linked boards",401);
 if(Number(req.headers.get("content-length")||0)>3000)return fail("Request too large",413);
 const input=await req.json().catch(()=>({}));
 const targets=Array.isArray(input.items)?input.items:[];
 if(!targets.length||targets.length>2)return fail("Send one or two discoveries per request",400);
 const requested=Array.isArray(input.fields)?input.fields:["price","image","details","description"];
 const allowed=new Set(["price","image","details","description"]);
 const fields=new Set(requested.filter(x=>allowed.has(x)));
 if(!fields.size)return fail("Select a field to improve",400);
 const rateKey="health:rate:"+user.id+":"+Math.floor(Date.now()/900000);
 const used=Number(await GATHERLY_KV.get(rateKey)||0);
 if(used+targets.length>80)return fail("Batch limit reached. Try again after the 15-minute window.",429);
 await GATHERLY_KV.put(rateKey,String(used+targets.length),{expirationTtl:960});
 const out=[];
 for(const entry of targets){
  const boardId=String(entry?.boardId||""),itemId=String(entry?.itemId||"");
  if(!G23_UUID.test(boardId)||!G23_UUID.test(itemId)){out.push({boardId,itemId,status:"invalid"});continue}
  const member=await GATHERLY_KV.get("ub:"+user.id+":"+boardId,"json");
  if(!member){out.push({boardId,itemId,status:"forbidden"});continue}
  const key="i:"+boardId+":"+itemId,old=await GATHERLY_KV.get(key,"json");
  if(!old){out.push({boardId,itemId,status:"missing"});continue}
  if(!v8SafeURL(old.url)){out.push({boardId,itemId,status:"no_source"});continue}
  let metadata;
  try{metadata=await g31Preview(await g26PublicPreview(old.url))}
  catch(error){out.push({boardId,itemId,status:"blocked",message:String(error.message||"Fetch failed").slice(0,100)});continue}
  if(!metadata?.fetched){out.push({boardId,itemId,status:"blocked",message:metadata?.notice||"Source did not provide public details"});continue}
  const item={...old},changes=[];
  const present=x=>String(x??"").trim().length>0;
  if(fields.has("price")&&!present(old.price)&&metadata.priceStatus==="found"&&present(metadata.price)&&metadata.currency==="TND"){
   item.price=metadata.price;item.currency="TND";
   if(metadata.priceOrigin)item.priceOrigin=g31Origin(metadata.priceOrigin);
   const hist=proHistoryEntry(item);if(hist)item.priceHistory=[...(Array.isArray(old.priceHistory)?old.priceHistory:[]),hist].slice(-40);
   changes.push("price")
  }
  if(fields.has("image")&&!present(old.image)&&present(metadata.image)&&v8SafeURL(metadata.image)){
   item.image=metadata.image;
   item.images=v15Images([metadata.image,...(Array.isArray(old.images)?old.images:[])],metadata.image);
   changes.push("image")
  }
  if(fields.has("description")&&!present(old.notes)&&present(metadata.description)){
   item.notes=String(metadata.description).slice(0,600);changes.push("description")
  }
  if(fields.has("details")&&metadata.structuredCompare){
   const base=old.compare&&typeof old.compare==="object"?old.compare:{};
   const next={...base},from=metadata.structuredCompare;
   for(const k of ["brand","model","availability","condition"]){
    if(!present(next[k])&&present(from[k])){next[k]=String(from[k]).slice(0,180);changes.push(k)}
   }
   const specs=Array.isArray(next.specs)?[...next.specs]:[],existing=new Set(specs.map(x=>String(x.name||"").trim().toLowerCase()));
   for(const sp of Array.isArray(from.specs)?from.specs:[]){
    const name=String(sp.name||"").trim(),value=String(sp.value||"").trim();
    if(name&&value&&!existing.has(name.toLowerCase())&&specs.length<24){
     specs.push({name:name.slice(0,65),value:value.slice(0,180)});existing.add(name.toLowerCase());changes.push("specification")
    }
   }
   if(changes.some(x=>["brand","model","availability","condition","specification"].includes(x)))item.compare=g30CompareSafe({...next,specs})
  }
  if(changes.length){
   await GATHERLY_KV.put(key,JSON.stringify(item));
   const board=await GATHERLY_KV.get("b:"+boardId,"json");
   if(board)await g25BoardActivity(boardId,board,{type:"item_updated",itemId,itemTitle:item.title,collectionId:item.collectionId||"default"},user);
  }
  out.push({boardId,itemId,status:changes.length?"updated":"unchanged",changed:[...new Set(changes)],price:changes.includes("price")?item.price:undefined,currency:changes.includes("price")?"TND":undefined})
 }
 return authJson({results:out});
}
