 const m=String(raw||"").match(/^(\d{4})-(0[1-9]|1[0-2])$/);
 return m&&Number(m[1])>=2020&&Number(m[1])<=2100?m[0]:null;
}
function g20CartPrice(raw){
 if(raw===null||raw===undefined||raw==="")return null;
 let value=String(raw).trim().replace(/[^\d,.]/g,"");
 if(!value)return null;
 const comma=value.lastIndexOf(","),dot=value.lastIndexOf(".");
 if(comma>dot)value=value.replace(/\./g,"").replace(",",".");
 else if(dot>comma)value=value.replace(/,/g,"");
 else value=value.replace(",",".");
 const n=Number(value);
 return Number.isFinite(n)&&n>=0&&n<=1000000000?Math.round(n*100)/100:null;
}
function g20CartCurrency(raw){
 const c=String(raw||"TND").trim().toUpperCase();
 return /^[A-Z]{3}$/.test(c)?c:"TND";
}
function g20CartUnit(raw){
 const n=Number(raw);
 return Number.isInteger(n)&&n>=1&&n<=999?n:null;
}
