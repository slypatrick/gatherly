/* Gatherly V15 — stream the beginning of large pages, rather than rejecting them */
async function v15ReadPreview(resp,max=1000000){
 const reader=resp.body?.getReader();
 if(!reader)throw Error("Source is empty");
 const decoder=new TextDecoder("utf-8");
 let text="",total=0;
 try{
  while(total<max){
   const {value,done}=await reader.read();
   if(done)break;
   if(!value?.length)continue;
   const bytes=value.subarray(0,Math.min(value.length,max-total));
   total+=bytes.length;
   text+=decoder.decode(bytes,{stream:true});
   if(value.length>bytes.length)break;
  }
 }finally{reader.cancel().catch(()=>{})}
 return text+decoder.decode();
}
